Actualmente el mapa usa Leaflet con tiles de OpenStreetMap directamente, lo cual no es apto para producción por los límites de uso de OSM. Necesito migrar a Mapbox GL JS que tiene 50,000 cargas gratuitas al mes y se ve mucho mejor.

SETUP
Desinstala las dependencias anteriores de Leaflet:
bashpnpm remove leaflet react-leaflet @types/leaflet --filter @workspace/mercanto
Instala Mapbox:
bashpnpm add mapbox-gl --filter @workspace/mercanto
pnpm add -D @types/mapbox-gl --filter @workspace/mercanto
Agrega al .env.example:
# Mapbox (mapa)
VITE_MAPBOX_TOKEN=
Agrega la variable VITE_MAPBOX_TOKEN en las variables de entorno de Replit con tu token de Mapbox. Para obtenerlo: ve a mapbox.com, crea cuenta gratuita, entra a Account → Tokens y copia el token público que empieza con pk..
En artifacts/mercanto/src/index.css o el CSS global, agrega el import del CSS de Mapbox:
css@import 'mapbox-gl/dist/mapbox-gl.css';

COMPONENTE 1 — Mapa de tiendas (página /map o StoreMap)
Busca el componente que muestra el mapa con los pins de las tiendas (probablemente src/components/shared/StoreMap.tsx o src/pages/map.tsx). Reemplaza toda la implementación de Leaflet por esta de Mapbox:
tsximport { useEffect, useRef } from "react";
import mapboxgl from "mapbox-gl";

mapboxgl.accessToken = import.meta.env.VITE_MAPBOX_TOKEN;

interface StoreMapProps {
  stores: Array<{
    id: number;
    name: string;
    slug: string;
    lat: string | number | null;
    lng: string | number | null;
    logoUrl?: string | null;
    district?: string | null;
    whatsapp?: string | null;
    category?: { name: string; icon?: string | null } | null;
  }>;
  centerLat?: number;
  centerLng?: number;
  zoom?: number;
}

export function StoreMap({ stores, centerLat = -11.1297, centerLng = -75.3500, zoom = 13 }: StoreMapProps) {
  const mapContainer = useRef<HTMLDivElement>(null);
  const map = useRef<mapboxgl.Map | null>(null);
  const markers = useRef<mapboxgl.Marker[]>([]);

  useEffect(() => {
    if (!mapContainer.current || map.current) return;

    map.current = new mapboxgl.Map({
      container: mapContainer.current,
      style: "mapbox://styles/mapbox/streets-v12",
      center: [centerLng, centerLat],
      zoom,
    });

    map.current.addControl(new mapboxgl.NavigationControl(), "top-right");
    map.current.addControl(new mapboxgl.GeolocateControl({
      positionOptions: { enableHighAccuracy: true },
      trackUserLocation: false,
      showUserHeading: false,
    }), "top-right");

    return () => {
      markers.current.forEach(m => m.remove());
      markers.current = [];
      map.current?.remove();
      map.current = null;
    };
  }, []);

  useEffect(() => {
    if (!map.current) return;

    markers.current.forEach(m => m.remove());
    markers.current = [];

    stores.forEach(store => {
      const lat = parseFloat(String(store.lat));
      const lng = parseFloat(String(store.lng));
      if (isNaN(lat) || isNaN(lng)) return;

      const el = document.createElement("div");
      el.className = "store-marker";
      el.innerHTML = `
        <div style="
          background:#2563EB;
          color:#fff;
          border-radius:50%;
          width:36px;height:36px;
          display:flex;align-items:center;justify-content:center;
          font-size:16px;
          box-shadow:0 2px 8px rgba(0,0,0,0.25);
          border:2px solid #fff;
          cursor:pointer;
        ">${store.category?.icon || "🏪"}</div>
      `;

      const popup = new mapboxgl.Popup({ offset: 25, closeButton: false, maxWidth: "220px" })
        .setHTML(`
          <div style="font-family:sans-serif;padding:4px">
            <p style="font-weight:600;font-size:14px;margin:0 0 2px">${store.name}</p>
            <p style="font-size:12px;color:#666;margin:0 0 6px">${store.district || ""} ${store.category ? "· " + store.category.name : ""}</p>
            <div style="display:flex;gap:6px">
              <a href="/stores/${store.slug}" style="font-size:12px;color:#2563EB;text-decoration:none;font-weight:500">Ver tienda →</a>
              ${store.whatsapp ? `<a href="https://wa.me/${store.whatsapp}" target="_blank" style="font-size:12px;color:#16a34a;text-decoration:none;font-weight:500">WhatsApp</a>` : ""}
            </div>
          </div>
        `);

      const marker = new mapboxgl.Marker(el)
        .setLngLat([lng, lat])
        .setPopup(popup)
        .addTo(map.current!);

      markers.current.push(marker);
    });

    if (stores.length > 0) {
      const validStores = stores.filter(s => !isNaN(parseFloat(String(s.lat))) && !isNaN(parseFloat(String(s.lng))));
      if (validStores.length > 1) {
        const bounds = new mapboxgl.LngLatBounds();
        validStores.forEach(s => bounds.extend([parseFloat(String(s.lng)), parseFloat(String(s.lat))]));
        map.current.fitBounds(bounds, { padding: 60, maxZoom: 15 });
      } else if (validStores.length === 1) {
        map.current.flyTo({ center: [parseFloat(String(validStores[0].lng)), parseFloat(String(validStores[0].lat))], zoom: 15 });
      }
    }
  }, [stores]);

  return <div ref={mapContainer} style={{ width: "100%", height: "100%", minHeight: "400px", borderRadius: "12px" }} />;
}

COMPONENTE 2 — Mini mapa para seleccionar ubicación al crear tienda
En artifacts/mercanto/src/pages/create-store.tsx, reemplaza toda la lógica del mini mapa de Leaflet por esta implementación con Mapbox. Busca donde está miniMapRef, miniMapInstanceRef, miniMarkerRef y reemplaza la inicialización del mapa:
tsximport mapboxgl from "mapbox-gl";
mapboxgl.accessToken = import.meta.env.VITE_MAPBOX_TOKEN;

// Dentro del useEffect que inicializa el mini mapa:
useEffect(() => {
  if (!miniMapRef.current || miniMapInstanceRef.current) return;

  const initialCenter = coords
    ? [coords.lng, coords.lat]
    : [SAN_RAMON_CENTER.lng, SAN_RAMON_CENTER.lat];

  miniMapInstanceRef.current = new mapboxgl.Map({
    container: miniMapRef.current,
    style: "mapbox://styles/mapbox/streets-v12",
    center: initialCenter as [number, number],
    zoom: 15,
  });

  miniMapInstanceRef.current.addControl(new mapboxgl.NavigationControl(), "top-right");

  const marker = new mapboxgl.Marker({ draggable: true, color: "#f97316" })
    .setLngLat(initialCenter as [number, number])
    .addTo(miniMapInstanceRef.current);

  miniMarkerRef.current = marker;

  marker.on("dragend", () => {
    const lngLat = marker.getLngLat();
    setCoords({ lat: lngLat.lat, lng: lngLat.lng });
  });

  miniMapInstanceRef.current.on("click", (e: mapboxgl.MapMouseEvent) => {
    marker.setLngLat(e.lngLat);
    setCoords({ lat: e.lngLat.lat, lng: e.lngLat.lng });
  });

  return () => {
    miniMapInstanceRef.current?.remove();
    miniMapInstanceRef.current = null;
  };
}, [miniMapRef.current]);

// Cuando el usuario usa GPS y cambia coords, mueve el marker:
useEffect(() => {
  if (!coords || !miniMapInstanceRef.current || !miniMarkerRef.current) return;
  miniMarkerRef.current.setLngLat([coords.lng, coords.lat]);
  miniMapInstanceRef.current.flyTo({ center: [coords.lng, coords.lat], zoom: 16 });
}, [coords]);
